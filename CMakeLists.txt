cmake_minimum_required(VERSION 3.16)
project(recon VERSION 2.1.0 DESCRIPTION "TCP/UDP port, service and OS scanner" LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "recon uses epoll and Linux socket semantics; Linux is the only supported platform.")
endif()

# ---- options -----------------------------------------------------------------------------
set(RECON_CXX_STANDARD "17" CACHE STRING "C++ standard: 17 (std::thread wrapper) or 20 (std::jthread)")
set_property(CACHE RECON_CXX_STANDARD PROPERTY STRINGS 17 20)
option(RECON_BUILD_TESTS "Build the test suite" ON)
option(RECON_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(RECON_HARDENING "Build with security hardening flags (bounds-checked containers, fortify, stack protector, PIE, RELRO)" ON)
option(RECON_BUILD_FUZZERS "Build fuzz targets (libFuzzer with Clang, otherwise a standalone driver)" OFF)
set(RECON_WITH_OPENSSL "AUTO" CACHE STRING "TLS certificate inspection: AUTO, ON or OFF")
set_property(CACHE RECON_WITH_OPENSSL PROPERTY STRINGS AUTO ON OFF)
set(RECON_SANITIZER "" CACHE STRING "Sanitizers: 'address;undefined' or 'thread' (Debug builds)")

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_CXX_STANDARD ${RECON_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)  # consumed by clang-tidy

find_package(Threads REQUIRED)

set(RECON_HAVE_OPENSSL OFF)
if(NOT RECON_WITH_OPENSSL STREQUAL "OFF")
  if(RECON_WITH_OPENSSL STREQUAL "ON")
    find_package(OpenSSL REQUIRED)
  else()
    find_package(OpenSSL QUIET)
  endif()
  if(OpenSSL_FOUND)
    set(RECON_HAVE_OPENSSL ON)
  endif()
endif()
message(STATUS "recon: C++${RECON_CXX_STANDARD}, OpenSSL TLS inspection: ${RECON_HAVE_OPENSSL}")

# ---- hardening: only flags the compiler actually supports are used -------------------------------
include(CheckCXXCompilerFlag)
set(RECON_HARDEN_COMPILE "")
set(RECON_HARDEN_LINK "")
if(RECON_HARDENING AND NOT RECON_SANITIZER)
  foreach(flag -fstack-protector-strong -fstack-clash-protection -fcf-protection=full)
    string(MAKE_C_IDENTIFIER "RECON_HAS${flag}" _var)
    check_cxx_compiler_flag(${flag} ${_var})
    if(${_var})
      list(APPEND RECON_HARDEN_COMPILE ${flag})
    endif()
  endforeach()
  # Bounds-checked std::vector/string/array in libstdc++, and fortified libc calls (needs optimisation).
  list(APPEND RECON_HARDEN_COMPILE -D_GLIBCXX_ASSERTIONS)
  if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    list(APPEND RECON_HARDEN_COMPILE -U_FORTIFY_SOURCE -D_FORTIFY_SOURCE=2)
  endif()
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  list(APPEND RECON_HARDEN_LINK -pie -Wl,-z,relro,-z,now,-z,noexecstack)
endif()

# ---- flags -------------------------------------------------------------------------------
function(recon_configure target)
  target_compile_options(${target} PRIVATE ${RECON_HARDEN_COMPILE})
  target_link_options(${target} PRIVATE ${RECON_HARDEN_LINK})
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wnon-virtual-dtor
                                           -Wold-style-cast -Wcast-align -Wformat=2)
  if(RECON_WARNINGS_AS_ERRORS)
    target_compile_options(${target} PRIVATE -Werror)
  endif()
  if(RECON_SANITIZER)
    string(REPLACE ";" "," _san "${RECON_SANITIZER}")
    target_compile_options(${target} PRIVATE -fsanitize=${_san} -fno-omit-frame-pointer -fno-sanitize-recover=all -g)
    target_link_options(${target} PRIVATE -fsanitize=${_san})
  endif()
endfunction()

# ---- library -----------------------------------------------------------------------------
add_library(recon_core STATIC
  src/net.cpp src/synscan.cpp src/json.cpp src/baseline.cpp src/target.cpp src/scanner.cpp src/matchers.cpp src/detector.cpp
  src/report.cpp src/osdetect.cpp src/options.cpp src/scan.cpp
  src/fileio.cpp)
target_include_directories(recon_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
                                       PRIVATE src)
target_link_libraries(recon_core PUBLIC Threads::Threads)
if(RECON_HAVE_OPENSSL)
  target_sources(recon_core PRIVATE src/tls_openssl.cpp)
  target_compile_definitions(recon_core PUBLIC RECON_WITH_OPENSSL=1)
  target_link_libraries(recon_core PUBLIC OpenSSL::SSL OpenSSL::Crypto)
endif()
recon_configure(recon_core)

add_executable(recon src/main.cpp)
target_link_libraries(recon PRIVATE recon_core)
recon_configure(recon)
install(TARGETS recon RUNTIME DESTINATION bin)
install(FILES README.md LICENSE docs/MANUAL.md docs/ARCHITECTURE.md DESTINATION share/doc/recon)
install(FILES man/recon.1 DESTINATION share/man/man1)
install(PROGRAMS scripts/cert_expiry.py scripts/drift_cron.sh DESTINATION share/recon/scripts)
install(FILES completions/recon.bash DESTINATION share/bash-completion/completions RENAME recon)
install(FILES completions/_recon DESTINATION share/zsh/site-functions)

# ---- tests -------------------------------------------------------------------------------
if(RECON_BUILD_TESTS)
  enable_testing()
  set(_unit target matchers report infra json)
  set(_integration net scanner detector syn)  # open real sockets; timing sensitive
  foreach(t IN LISTS _unit _integration)
    add_executable(test_${t} tests/test_main.cpp tests/test_${t}.cpp)
    target_link_libraries(test_${t} PRIVATE recon_core)
    target_include_directories(test_${t} PRIVATE tests)
    recon_configure(test_${t})
    add_test(NAME ${t} COMMAND test_${t})
    set_tests_properties(${t} PROPERTIES TIMEOUT 300)
  endforeach()
  foreach(t IN LISTS _unit)
    set_tests_properties(${t} PROPERTIES LABELS unit)
  endforeach()
  foreach(t IN LISTS _integration)
    # Serialise: these assert on wall-clock behaviour and must not compete for CPU.
    set_tests_properties(${t} PROPERTIES LABELS integration RESOURCE_LOCK timing)
  endforeach()

  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli_e2e COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_cli.py $<TARGET_FILE:recon>)
    set_tests_properties(cli_e2e PROPERTIES LABELS e2e TIMEOUT 300 RESOURCE_LOCK timing)
    add_test(NAME manual COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/doc_test.py $<TARGET_FILE:recon>
             ${CMAKE_CURRENT_SOURCE_DIR}/docs/MANUAL.md ${CMAKE_CURRENT_SOURCE_DIR}/man/recon.1)
    set_tests_properties(manual PROPERTIES LABELS docs TIMEOUT 120)
  else()
    message(STATUS "recon: Python3 not found; end-to-end CLI tests disabled")
  endif()
endif()

# ---- fuzzing -----------------------------------------------------------------------------
if(RECON_BUILD_FUZZERS)
  foreach(t matchers json inputs)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      add_executable(fuzz_${t} fuzz/fuzz_${t}.cpp)
      target_compile_options(fuzz_${t} PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
      target_link_options(fuzz_${t} PRIVATE -fsanitize=fuzzer,address,undefined)
    else()
      add_executable(fuzz_${t} fuzz/fuzz_${t}.cpp fuzz/standalone_driver.cpp)
    endif()
    target_link_libraries(fuzz_${t} PRIVATE recon_core)
    add_test(NAME fuzz_${t}_smoke COMMAND fuzz_${t} ${CMAKE_CURRENT_SOURCE_DIR}/fuzz/corpus/${t} -runs=20000)
    set_tests_properties(fuzz_${t}_smoke PROPERTIES LABELS fuzz TIMEOUT 120)
  endforeach()
endif()
